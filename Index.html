============================================================================ $ErrorActionPreference = "Stop" $Api = "https://api.github.com" $HeadersBase = @{ Accept = "application/vnd.github+json" "X-GitHub-Api-Version" = "2022-11-28" "User-Agent" = "Vyomaraj-Sync-Diagnostic" } function Convert-SecureStringToPlainText { param([Security.SecureString]$SecureString) $ptr = [Runtime.InteropServices.Marshal]::SecureStringToBSTR($SecureString) try { [Runtime.InteropServices.Marshal]::PtrToStringBSTR($ptr) } finally { [Runtime.InteropServices.Marshal]::ZeroFreeBSTR($ptr) } } function Get-HttpErrorDetails { param($ErrorRecord) $status = $null $body = $null try { $response = $ErrorRecord.Exception.Response if ($response) { try { $status = [int]$response.StatusCode } catch {} try { $stream = $response.GetResponseStream() if ($stream) { $reader = New-Object IO.StreamReader($stream) $body = $reader.ReadToEnd() $reader.Dispose() } } catch {} } } catch {} [pscustomobject]@{ HttpStatus = $status Message = $ErrorRecord.Exception.Message Response = $body } } function Invoke-GitHubApi { param( [Parameter(Mandatory)] [string]$Token, [Parameter(Mandatory)] [string]$Method, [Parameter(Mandatory)] [string]$Path, $Body ) $headers = $HeadersBase.Clone() $headers.Authorization = "Bearer $Token" $uri = "$Api$Path" if ($null -ne $Body) { return Invoke-RestMethod -Method $Method -Uri $uri -Headers $headers ` -ContentType "application/json" ` -Body ($Body | ConvertTo-Json -Depth 10) } return Invoke-RestMethod -Method $Method -Uri $uri -Headers $headers } function Test-Repository { param( [Parameter(Mandatory)] [string]$Owner, [Parameter(Mandatory)] [string]$Repo, [Parameter(Mandatory)] [string]$Branch, [Parameter(Mandatory)] [string]$Token ) $result = [ordered]@{ Repository = "$Owner/$Repo" Branch = $Branch AuthenticatedAs = $null DefaultBranch = $null MainHeadBefore = $null Read = "NOT_TESTED" WriteTest = "NOT_TESTED" RemoteHeadVerified = $false TestCommitSha = $null DiagnosticBranch = $null Cleanup = "NOT_REQUIRED" Errors = @() } $createdBranch = $false $diagnosticBranch = "sync-diagnostic-" + (Get-Date -Format "yyyyMMddHHmmss") + "-" + ([guid]::NewGuid().ToString("N").Substring(0, 8)) $result.DiagnosticBranch = $diagnosticBranch try { $user = Invoke-GitHubApi $Token GET "/user" $result.AuthenticatedAs = $user.login $repoInfo = Invoke-GitHubApi $Token GET "/repos/$Owner/$Repo" $result.DefaultBranch = $repoInfo.default_branch $branchInfo = Invoke-GitHubApi $Token GET ` "/repos/$Owner/$Repo/branches/$Branch" $baseSha = $branchInfo.commit.sha $result.MainHeadBefore = $baseSha $result.Read = "PASS" # Create an isolated test branch based on the requested branch. $null = Invoke-GitHubApi $Token POST ` "/repos/$Owner/$Repo/git/refs" @{ ref = "refs/heads/$diagnosticBranch" sha = $baseSha } $createdBranch = $true # A successful Contents API write tests actual repository write access. $filePath = ".vyomaraj-sync-diagnostic/$diagnosticBranch.txt" $text = "Temporary Vyomaraj/GitHub sync diagnostic. Safe to delete.`n" $content = [Convert]::ToBase64String( [Text.Encoding]::UTF8.GetBytes($text) ) $commit = Invoke-GitHubApi $Token PUT ` "/repos/$Owner/$Repo/contents/$filePath" @{ message = "Temporary sync diagnostic $diagnosticBranch" content = $content branch = $diagnosticBranch } $result.TestCommitSha = $commit.commit.sha # Confirm GitHub's remote branch ref points to the created commit. $remoteRef = Invoke-GitHubApi $Token GET ` "/repos/$Owner/$Repo/git/ref/heads/$diagnosticBranch" if ($remoteRef.object.sha -eq $result.TestCommitSha) { $result.RemoteHeadVerified = $true $result.WriteTest = "PASS" } else { $result.WriteTest = "FAIL" $result.Errors += "Remote branch HEAD did not match the returned commit SHA." } } catch { $details = Get-HttpErrorDetails $_ $result.Errors += [pscustomobject]@{ HttpStatus = $details.HttpStatus Message = $details.Message Response = $details.Response } if ($result.Read -ne "PASS") { $result.Read = "FAIL" } else { $result.WriteTest = "FAIL" } } finally { # Delete only the uniquely named diagnostic branch created by this run. if ($createdBranch) { try { $null = Invoke-GitHubApi $Token DELETE ` "/repos/$Owner/$Repo/git/refs/heads/$diagnosticBranch" $result.Cleanup = "DELETED" } catch { $details = Get-HttpErrorDetails $_ $result.Cleanup = "FAILED" $result.Errors += [pscustomobject]@{ CleanupFailure = $true HttpStatus = $details.HttpStatus Message = $details.Message Response = $details.Response } } } } [pscustomobject]$result } Write-Host "This test creates a temporary branch and diagnostic commit in each repository." Write-Host "It does NOT modify main. Tokens are not written to the report."
